cmake_minimum_required(VERSION 3.16)

set(IDF_TARGET "esp32s3")
set(ZONE_LITE_FAMILY "zkt" CACHE STRING "Terminal firmware family (zkt or hikvision)")
if(ZONE_LITE_FAMILY STREQUAL "hikvision")
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.hikvision.defaults")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(ZONE_LITE_FAMILY STREQUAL "hikvision")
project(zone_lite_hikvision VERSION 3.1.1)
elseif(ZONE_LITE_FAMILY STREQUAL "zkt")
project(zone_lite VERSION 2.6.8)
else()
    message(FATAL_ERROR "Unknown Zone Lite firmware family")
endif()
